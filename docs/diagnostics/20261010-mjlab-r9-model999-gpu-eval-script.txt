import json,sys,time
from pathlib import Path
sys.path.insert(0,str(Path.cwd()/'mjlab_training/src'))
import torch
from ct_mjlab.policy import load_actor
from ct_mjlab.reference_training.entry import RecoverVecEnv
checkpoint=Path('logs/mjlab/chuanliantui/20261010_101723_976809_wheelleg_recover_r9_no_knee_4096_1000/model_999.pt')
actor=load_actor(checkpoint,'cuda:0');results={}
for randomize in (False,True):
 env=RecoverVecEnv(num_envs=32,device='cuda:0',randomize=randomize,seed=42,auto_reset=False)
 raw=env.unwrapped;state=raw.command_manager.get_term('standup');robot=raw.scene['robot']
 active=torch.ones(32,dtype=torch.bool,device='cuda:0');recovered=torch.zeros_like(active);numerical=torch.zeros_like(active)
 failures=[None]*32;last=[None]*32
 try:
  for i in range(1000):
   with torch.inference_mode():a=actor(env.obs)
   _,_,done,_=env.step(a)
   d=robot.data;g=d.projected_gravity_b
   base=raw.scene['base_contact'].data.force.norm(dim=-1).amax(-1)
   rows=torch.stack((d.root_link_pos_w[:,2],state.command[:,2],torch.atan2(-g[:,1],-g[:,2]).abs()*180/torch.pi,torch.asin(g[:,0].clamp(-1,1)).abs()*180/torch.pi,base),-1).cpu().tolist()
   recovered |= state.strict_recovered & active;numerical |= state.numerical_failure & active
   for j in torch.nonzero(active).flatten().cpu().tolist():
    h,c,r,p,f=rows[j];last[j]=dict(time=(i+1)*raw.step_dt,height=h,command_height=c,roll_abs_deg=r,pitch_abs_deg=p,base_contact_force_n=f)
   for j in torch.nonzero(active & raw.reset_terminated).flatten().cpu().tolist():failures[j]=last[j]
   active &= ~done.bool()
   if not bool(active.any()):break
   if bool(done.any()):
    raw.reset(env_ids=torch.nonzero(done).flatten())
    env.obs=env.get_observations()
  summary=dict(num_envs=32,seed=42,randomization=randomize,requested_seconds=20,steps_executed=i+1,strict_recovered_envs=int(recovered.sum()),numerical_failure_envs=int(numerical.sum()),failure_envs=sum(x is not None for x in failures),surviving_envs=int(active.sum()),first_failure_per_env=failures,last_active_sample_per_env=last,mean_last_active_height=sum(x['height'] for x in last)/32,mean_last_active_command_height=sum(x['command_height'] for x in last)/32,autoreset=False,finished_envs_reset_then_excluded=True,policy='deterministic',learning=False)
  results['training_randomization' if randomize else 'nominal']=summary
  Path('docs/diagnostics/20261010-mjlab-r9-model999-gpu-eval.json').write_text(json.dumps(results,indent=2)+'\n')
  print('RESULT',json.dumps({k:v for k,v in summary.items() if k not in ('first_failure_per_env','last_active_sample_per_env')}),flush=True)
 finally:env.close()
