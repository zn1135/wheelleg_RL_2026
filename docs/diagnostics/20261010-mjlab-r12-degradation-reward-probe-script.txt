import json,sys,hashlib
from pathlib import Path
sys.path.insert(0,str(Path.cwd()/'mjlab_training/src'))
import torch
from ct_mjlab.policy import load_actor,checkpoint_reset_config,checkpoint_policy_joint_order
from ct_mjlab.reference_training.entry import RecoverVecEnv
report={'purpose':'Both actors under identical r12 rewards, deterministic32 nominal envs, seed42, no learning; physical failures excluded afterwards.','groups':{}}
for name in ('r11','r12'):
 checkpoint=Path(json.loads(Path('/tmp/ct-'+name+'-cloud-eval-path.json').read_text())['checkpoint']);actor=load_actor(checkpoint,'cuda:0')
 env=RecoverVecEnv(num_envs=32,device='cuda:0',randomize=False,seed=42,auto_reset=False,reset_config=checkpoint_reset_config(checkpoint),dynamic_modes=True,policy_joint_order=checkpoint_policy_joint_order(checkpoint),knee_limit_weight=0.)
 raw=env.unwrapped;raw.reset(seed=42);env.obs=env.get_observations();initial_hash=hashlib.sha256(raw.sim.data.qpos.cpu().numpy().tobytes()).hexdigest()
 initial=raw.sim.data.qpos.clone();initial_command=raw.command_manager.get_term('standup').command.clone()
 if name=='r11':reference_initial=initial;reference_command=initial_command
 initial_max_diff=float((initial-reference_initial).abs().max());command_max_diff=float((initial_command-reference_command).abs().max())
 active=torch.ones(32,device=raw.device,dtype=torch.bool);ret=torch.zeros(32,device=raw.device);disc=ret.clone();count=ret.clone();failure=[None]*32;positive=ret.clone();negative=ret.clone()
 per_term={name:ret.clone() for name in raw.reward_manager.active_terms};rolling=[]
 try:
  for i in range(1000):
   with torch.inference_mode():act=actor(env.obs)
   _,reward,done,_=env.step(act)
   ret+=reward*active;disc+=(.99**i)*reward*active;count+=active
   positive+=reward.clamp_min(0)*active;negative+=reward.clamp_max(0)*active
   for term,cfg in zip(raw.reward_manager.active_terms,raw.reward_manager._term_cfgs):
    if cfg.weight!=0:per_term[term]+=raw._huananhu_rewards.values[term]*cfg.weight*raw.step_dt*active
   if i>=500 and bool(active.any()):rolling.append(float(reward[active].mean()))
   for j in torch.nonzero(active&raw.reset_terminated).flatten().cpu().tolist():failure[j]=(i+1)*raw.step_dt
   active&=~done.bool()
   if not bool(active.any()):break
   if bool(done.any()):raw.reset(env_ids=torch.nonzero(done).flatten());env.obs=env.get_observations()
  report['groups'][name]=dict(checkpoint=str(checkpoint),initial_qpos_sha256=initial_hash,initial_qpos_max_abs_diff_from_r11=initial_max_diff,initial_command_max_abs_diff_from_r11=command_max_diff,mean_episode_seconds=float(count.mean())*raw.step_dt,physical_failures=sum(v is not None for v in failure),first_failure_s=failure,mean_undiscounted_return=float(ret.mean()),mean_discounted_return_gamma99=float(disc.mean()),mean_step_reward=float((ret/count).mean()),mean_last10s_active_step_reward=sum(rolling)/len(rolling) if rolling else None,mean_term_return={k:float(v.mean()) for k,v in per_term.items()})
  print(name,json.dumps(report['groups'][name]),flush=True)
 finally:env.close()
Path('/tmp/ct-r12-reward-probe.json').write_text(json.dumps(report,indent=2)+'\n')
