import json, torch
from ct_mjlab.environment import ChuanliantuiEnv
cases=[('baseline',None,False),('no_motor_torque',None,True),('tau005',.005,False),('tau004',.004,False)]
results=[]
for label,tau,zero in cases:
    env=ChuanliantuiEnv(4,'cuda:0',seed=42,randomize=True)
    if tau is not None:
        env.sim.expand_model_fields(('eq_solref',))
        env.sim.model.eq_solref[:,:,0]=tau
    if zero:
        original=env.recipe.torque
        def disabled(*a,**k):return torch.zeros_like(original(*a,**k))
        env.recipe.torque=disabled
    pairs=[(s+'_rear_pin'+p,s+'_front_pin'+p) for s in ('left','right') for p in ('1','2')]
    pairs += [(s+'_spring_tip',s+'_gas_spring_lower') for s in ('left','right')]
    siteids=[(env.sim.mj_model.site('robot/'+a).id,env.sim.mj_model.site('robot/'+b).id) for a,b in pairs]
    worst={'error_m':0};after={'error_m':0};per=[0.]*6;ticks=[0];numerical=0;ordinary=0
    step=env.sim.step;forward=env.sim.forward
    def tick(*a,**k):
        step(*a,**k);ticks[0]+=1
    def inspect(*a,**k):
        forward(*a,**k)
        for i,(first,second) in enumerate(siteids):
            err=torch.linalg.vector_norm(env.sim.data.site_xpos[:,first]-env.sim.data.site_xpos[:,second],dim=-1)
            value,idx=err.max(0);value=float(value);idx=int(idx);per[i]=max(per[i],value)
            item=dict(error_m=value,sites=pairs[i],env=idx,t_s=round(ticks[0]*.002,6),ground_start=bool(env.ground_start[idx]),root_z=float(env.robot.data.root_link_pos_w[idx,2]),motor_torque=env.sim.data.ctrl[idx,env.ctrl].tolist(),max_speed=float(env.sim.data.qvel[idx].abs().max()))
            if value>worst['error_m']:worst.update(item)
            if ticks[0]*.002>=.5 and value>after['error_m']:after.update(item)
    env.sim.step=tick;env.sim.forward=inspect
    for _ in range(128):
        obs,rew,done,_=env.step(torch.zeros(4,6,device='cuda:0'))
        numerical+=int(env.last_numerical_reset.sum());ordinary+=int(done.sum())
    result=dict(case=label,tau_s=tau,peak=worst,peak_after_05s=after,pair_max_m=per,numerical_resets=numerical,episode_resets=ordinary)
    print(json.dumps(result),flush=True);results.append(result);env.close()
open('/tmp/ct_closure_diagnose.json','w').write(json.dumps(results,indent=2))
