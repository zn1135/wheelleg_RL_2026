def check_dynamic_modes():
    mode=ControlMode(2,'cpu');mode.reset(torch.arange(2),torch.tensor([True,False]))
    target=torch.tensor([.30,.34]);tick=0
    state=dict(height=target.clone(),roll=torch.zeros(2),pitch=torch.zeros(2),
        base_contact=torch.zeros(2),wheel_contact=torch.full((2,2),6.),valid=torch.ones(2,dtype=torch.bool))
    def advance(count=1):
        nonlocal tick
        for _ in range(count):
            tick+=1;mode.advance(state,target,tick)
    assert torch.equal(mode.observation(target)[0],torch.tensor([0.,0,0,1,0,1.5,0]))
    advance(24);assert mode.recover[0]
    # Exactly 5 N is insufficient and resets the required contiguous hold.
    state['wheel_contact'][0,1]=5.;advance();assert mode.stable_steps[0]==0
    state['wheel_contact'][0,1]=6.;state['height'][0]=.22
    advance(25);assert mode.recover[0], 'old fixed .22 m was treated as standing'
    state['height'][0]=target[0];advance(24);assert mode.recover[0]
    advance();assert not mode.recover.any() and mode.recovered[0]
    assert torch.equal(mode.observation(target),torch.tensor([[1.,0,0,0,0,0,0]]).expand(2,-1))
    # Interrupted failures do not trigger recovery; five consecutive ticks do.
    state['base_contact'][1]=2.;advance(4);assert not mode.recover[1]
    state['base_contact'][1]=0.;advance();assert mode.failure_steps[1]==0
    state['base_contact'][1]=2.;advance(5)
    assert mode.recover[1] and mode.state_steps[1]==0
    advance();assert torch.isclose(mode.observation(target)[1,6],torch.tensor(.02))
    target[1]=.27
    assert torch.isclose(mode.observation(target)[1,5],torch.tensor(1.35))
    state['base_contact'][1]=0.;state['height'][1]=.34;advance(25);assert mode.recover[1]
    state['height'][1]=.27;state['valid'][1]=False;advance(25);assert mode.recover[1]
    state['valid'][1]=True;advance(25);assert not mode.recover[1]
    state['height'][1]=.159;advance(5);assert mode.recover[1]
    # Queries and repeated advancement at the same transition have no effect.
    slots=mode.observation(target).clone();mode.advance(state,target,tick)
    assert torch.equal(slots,mode.observation(target))
    mode.reset(torch.tensor([0]),torch.tensor([True]))
    assert torch.equal(slots[1],mode.observation(target)[1])
    assert mode.state_steps[0]==0 and not mode.recovered[0]
    return dict(recover_on_ground_reset=True,stable_hold_steps=25,failure_hold_steps=5,
        target_follows_current_height_command=True,actor_critic_equal=True,
        partial_reset_preserved=True,queries_do_not_advance=True,duplicate_transition_ignored=True)

def check_action_order():
    expected_names=('lf0','lf00','rf0','rf00','lfwheel','rfwheel')
    assert action.policy_joint_order==expected_names
    assert action.policy_to_physical.tolist()==[0,1,4,2,3,5]
    assert sensor.raw_action_order.tolist()==list(range(6))
    manager=raw.action_manager
    tensors=[raw.sim.data.qpos,raw.sim.data.qvel,raw.sim.data.ctrl,
        raw.episode_length_buf,action._raw_actions,action._control_actions,
        action._torques,action._action_history,action._entity.data.joint_effort_target,
        manager.action,manager.prev_action,manager.prev_prev_action]
    saved=[x.clone() for x in tensors];head=action._history_head
    had_termination=hasattr(raw,'reset_terminated')
    if not had_termination:raw.reset_terminated=torch.zeros(a.num_envs,dtype=torch.bool,device=raw.device)
    ctrl_ids=[model.actuator('robot/'+name+'_motor').id for name in expected_names]
    try:
        # No integration of the synthetic motor state: isolate six output slots
        # and inspect controls by actuator NAME, then restore every buffer.
        raw.episode_length_buf.fill_(action._ground_hold_steps)
        raw.sim.data.qpos[:,action._motor_q_adr]=action.nominal_positions
        raw.sim.data.qvel[:,action._motor_v_adr]=0.
        for slot,name in enumerate(expected_names):
            requested=torch.zeros(a.num_envs,6,device=raw.device);requested[:,slot]=.17
            action.process_actions(requested)
            action._action_history.copy_(action._control_actions.unsqueeze(0))
            action.apply_actions();raw.scene.write_data_to_sim()
            expected=torch.zeros_like(requested)
            physical_slot=('lf0','lf00','lfwheel','rf0','rf00','rfwheel').index(name)
            if slot<4:
                gain=action.cfg.leg_stiffness*action.kp_scale[:,physical_slot]*action.cfg.leg_position_scale
            else:
                gain=action.cfg.wheel_stiffness*action.kd_scale[:,physical_slot]*action.cfg.wheel_velocity_scale
            expected[:,slot]=.17*gain*action.motor_strength[:,physical_slot]
            assert torch.allclose(raw.sim.data.ctrl[:,ctrl_ids],expected,atol=1e-5),name
            assert torch.equal(sensor.basic(sensor.raw_sensors())[:,22:28],requested),name
        # Three unequal histories catch a permutation applied to only one frame.
        manager.action.copy_(torch.tensor([.1,.2,.3,.4,.5,.6],device=raw.device))
        manager.prev_action.copy_(torch.tensor([.3,-.1,.5,-.4,.2,-.6],device=raw.device))
        manager.prev_prev_action.copy_(torch.tensor([-.2,.1,-.3,.4,-.5,.6],device=raw.device))
        values=local.RewardState(raw).compute(raw)
        difference=manager.action-2*manager.prev_action+manager.prev_prev_action
        assert torch.allclose(values['action_smoothness_leg'],difference[:,:4].square().sum(-1))
        assert torch.allclose(values['action_smoothness_wheel'],difference[:,4:].square().sum(-1))
    finally:
        for tensor,previous in zip(tensors,saved):tensor.copy_(previous)
        action._history_head=head
        if not had_termination:del raw.reset_terminated
    return dict(policy_joint_order=list(expected_names),policy_to_physical=[0,1,4,2,3,5],
        six_named_actuator_pulses=True,requested_history_observation_identity=True,
        three_frame_leg_wheel_reward_mapping=True)
