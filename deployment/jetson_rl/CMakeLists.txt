cmake_minimum_required(VERSION 3.16)
project(jetson_rl LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -fno-fast-math)
set(PROTOCOL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../H7_RL_ros/imcalib/common")
option(JETSON_TENSORRT "Build Jetson TensorRT executable" ON)
if(JETSON_TENSORRT)
  find_package(CUDAToolkit REQUIRED)
  find_path(TENSORRT_INCLUDE NvInfer.h PATH_SUFFIXES aarch64-linux-gnu REQUIRED)
  find_library(TENSORRT nvinfer REQUIRED)
  find_library(ONNX_PARSER nvonnxparser REQUIRED)
  find_package(Threads REQUIRED)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/policy.hpp" OBS_HASH)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/model.hpp" MODEL_HASH)
  file(SHA256 "${PROTOCOL_DIR}/jetson_protocol.h" PROTOCOL_HASH)
  string(SHA256 CONFIG_HASH "${OBS_HASH};${MODEL_HASH};${PROTOCOL_HASH}")
  add_executable(jetson_rl src/main.cpp src/inference.cpp)
  target_compile_definitions(jetson_rl PRIVATE JRL_CONFIG_SHA="${CONFIG_HASH}")
  target_include_directories(jetson_rl PRIVATE ${PROTOCOL_DIR} ${TENSORRT_INCLUDE})
  target_link_libraries(jetson_rl PRIVATE ${TENSORRT} ${ONNX_PARSER} CUDA::cudart Threads::Threads)
endif()
add_executable(jetson_core_check ../../scripts/agent/jetson_rl/check_core.cpp)
target_include_directories(jetson_core_check PRIVATE ${PROTOCOL_DIR} src)

target_compile_options(jetson_core_check PRIVATE -UNDEBUG)
